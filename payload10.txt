    const next=!el.classList.contains('flipped');
    el.classList.toggle('flipped',next);
    el.setAttribute('aria-pressed',String(next));
  };

  el.addEventListener('keydown',e=>{
    if(e.key==='Enter' || e.key===' '){
      e.preventDefault();
      toggleFlip();
    }
  });
