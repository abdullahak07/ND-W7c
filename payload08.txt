const security=qs('security'), performance=qs('performance'), cost=qs('cost');
const sv=qs('sv'), pv=qs('pv'), cv=qs('cv'), fit=qs('fit'), ring=qs('ring'), tradeNote=qs('tradeNote');
function updateTrade(){
  const s=Number(security.value), p=Number(performance.value), c=Number(cost.value);
  sv.textContent=s; pv.textContent=p; cv.textContent=c;
  const balance=(100-Math.abs(s-p))*0.25 + (100-Math.abs(s-c))*0.20 + (100-Math.abs(p-c))*0.15;
  const raw=(s*0.40)+(p*0.25)+((100-c)*0.20)+balance;
  const score=Math.max(20,Math.min(100,Math.round(raw/1.15)));
  fit.textContent=score+'%';
  ring.style.setProperty('--deg',(score*3.6)+'deg');
  if(s>82 && p>80){tradeNote.textContent='High security + high performance usually means more cost, more area or more energy.';}
  else if(c>80){tradeNote.textContent='Very cost-sensitive designs tend to push toward smaller or serial implementations and may sacrifice speed.';}
  else if(s<40){tradeNote.textContent='If security priority is too low, the design may fit the hardware but not the threat model.';}
  else {tradeNote.textContent='Balanced profile: moderate resource budget with solid security.';}
}
[security,performance,cost].forEach(el=>el.addEventListener('input',updateTrade));
updateTrade();
